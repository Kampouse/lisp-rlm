(+ a b ...) — Add numbers. Single arg: identity. No args: 0.
(- a b ...) — Subtract. (- x) negates.
(* a b ...) — Multiply. Single arg: identity. No args: 1.
(/ a b ...) — Divide. Returns float if any arg is float.
(mod a b) — Modulo.
(abs n) — Absolute value.
(sqrt n) — Square root (returns float).
(expt base exp) — Exponentiation.
(equal? a b) — Deep structural equality.
(list a b ...) — Create a list.
(car lst) — First element of list.
(cdr lst) — Rest of list.
(cons head tail) — Prepend to list.
(append l1 l2) — Concatenate lists.
(reverse lst) — Reverse a list.
(sort lst) — Sort numbers (ascending) or strings (lexicographic).
(zip l1 l2) — Zip two lists into pairs.
(map f lst) — Apply f to each element, return new list.
(filter pred lst) — Keep elements matching predicate.
(reduce f init lst) — Fold left: f(accumulator, element).
(find pred lst) — First element matching predicate, or nil.
(some pred lst) — True if any element satisfies predicate.
(every pred lst) — True if all elements satisfy predicate.
(for-each f lst) — Apply f to each element for side effects. Returns nil.
(fold-left f init lst) — Left fold.
(fold-right f init lst) — Right fold.
(str-split s delim) — Split string by delimiter.
(str-trim s) — Trim whitespace.
(str-join sep lst) — Join list of strings with separator. Separator FIRST.
(number? x) — Is x a number?
(string? x) — Is x a string?
(map? x) — Is x a dict?
(procedure? x) — Is x callable (lambda)?
(symbol? x) — Is x a symbol?
(zero? n) — Is n zero?
(positive? n) — Is n > 0?
(negative? n) — Is n < 0?
(even? n) — Is n even?
(odd? n) — Is n odd?
(empty? x) — Is list/string empty?
(dict k1 v1 k2 v2 ...) — Create a dict.
(dict/get d key) — Get value from dict. Returns nil if missing.
(dict/set d key val) — Set key in dict (returns new dict).
(dict/has? d key) — Check if key exists.
(dict/keys d) — List of keys.
(dict/vals d) — List of values.
(json-set json key encoded-value) — Set/replace top-level key in JSON object, returns new JSON.
(print x) — Print value without newline.
(println x) — Print value with newline.
(read-file path) — Read file contents as string.
(write-file path content) — Write string to file.
(load-file path) — Load and evaluate a Lisp file.
(define name val) or (define (f params...) body) — Define binding.
(set! name val) — Mutate existing binding.
(lambda (params...) body) — Create anonymous function.
(if cond then else?) — Conditional.
(cond (test expr) ... (else expr)) — Multi-branch conditional.
(begin e1 e2 ...) — Sequence expressions, return last.
(let ((var val) ...) body) — Local bindings.
(let* ((var val) ...) body) — Sequential local bindings.
(letrec ((var val) ...) body) — Recursive local bindings.
(and e1 e2 ...) — Short-circuit logical AND.
(or e1 e2 ...) — Short-circuit logical OR.
(when cond body...) — Execute body if cond is truthy.
(unless cond body...) — Execute body if cond is falsy.
(apply f arg1 ... arglist) — Apply function to spread args.
(eval expr) — Evaluate a Lisp expression.
(error msg) — Raise an error.
(type-of x) — Return type name as string.
(infer-type f) — Probe a pure lambda to infer its type signature.
(pure-type f) — Return the pure type annotation, or nil.
(now) — Current Unix timestamp (milliseconds).
(elapsed since) — Milliseconds since given timestamp.
(sleep ms) — Sleep for milliseconds.
(save-state path val) — Serialize value to JSON file.
(load-state path) — Load value from JSON file.
